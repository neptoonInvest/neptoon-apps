function drawHistory(rows){ const canvas = document.getElementById("historyCanvas"); if(!canvas) return; const box = canvas.parentElement; const dpr = window.devicePixelRatio || 1; /* ========================= VALID HISTORY Portfolio + benchmark ========================= */ const history = (Array.isArray(rows) ? rows : []) .map(row => ({ date: row.date || "", portfolio: row.accPortIndex == null || row.accPortIndex === "" ? NaN : Number(row.accPortIndex), benchmark: row.accBenchIndex == null || row.accBenchIndex === "" ? NaN : Number(row.accBenchIndex) })) .filter(row => Number.isFinite(row.portfolio) || Number.isFinite(row.benchmark) ); /* ========================= CANVAS ========================= */ const w = Math.max( 320, box.clientWidth - 28 ); const h = Math.max( 220, box.clientHeight - 16 ); canvas.width = w * dpr; canvas.height = h * dpr; canvas.style.width = w + "px"; canvas.style.height = h + "px"; const ctx = canvas.getContext("2d"); ctx.setTransform( dpr, 0, 0, dpr, 0, 0 ); ctx.clearRect( 0, 0, w, h ); /* ========================= EMPTY HISTORY ========================= */ if(history.length < 2){ // Remove previous portfolio interactions canvas.onmousemove = null; canvas.onmouseleave = null; const tooltip = document.getElementById("historyTooltip"); if(tooltip){ tooltip.style.display = "none"; } ctx.fillStyle = "#64748b"; ctx.font = '11px Arial'; ctx.textAlign = "left"; ctx.fillText( INIT.lang === "da" ? "Ingen historiske afkast tilgængelige på 'Bankindestående'" : "No historical performance available on 'Bank Deposits' ", 16, 28 ); return; } /* ========================= COMBINED VALUES Both series determine the Y-axis range. ========================= */ const vals = history.flatMap(row => [ row.portfolio, row.benchmark ]).filter(Number.isFinite); /* ========================= CHART PADDING ========================= */ const pad = { l:48, r:14, t:36, b:30 }; /* ========================= Y AXIS Include index 100. Allow observations below 100. ========================= */ const dataMin = Math.min(...vals); const dataMax = Math.max(...vals); const rawMin = Math.min( 100, dataMin ); const rawMax = Math.max( 100, dataMax ); const rawRange = Math.max( 20, rawMax - rawMin ); const rawStep = rawRange / 4; const tickStep = getNiceHistoryTickStep( rawStep ); let axisMin = Math.floor( rawMin / tickStep ) * tickStep; let axisMax = Math.ceil( rawMax / tickStep ) * tickStep; if( dataMin >= 0 && axisMin < 0 ){ axisMin = 0; } if(axisMax <= axisMin){ axisMax = axisMin + tickStep; } /* ========================= COORDINATES ========================= */ const plotW = w - pad.l - pad.r; const plotH = h - pad.t - pad.b; const x = index => pad.l + index * plotW / (history.length - 1); const y = value => pad.t + (axisMax - value) * plotH / (axisMax - axisMin); /* ========================= GRID + Y AXIS ========================= */ ctx.strokeStyle = "#e4e8ee"; ctx.lineWidth = 1; const tickCount = Math.round( (axisMax - axisMin) / tickStep ); for( let i = 0; i <= tickCount; i++ ){ const axisValue = axisMin + i * tickStep; const yy = y(axisValue); ctx.beginPath(); ctx.moveTo( pad.l, yy ); ctx.lineTo( w - pad.r, yy ); ctx.stroke(); ctx.fillStyle = "#64748b"; ctx.font = '10px "Times New Roman"'; ctx.textAlign = "right"; ctx.fillText( axisValue.toFixed(0), pad.l - 7, yy + 3 ); } /* ========================= DRAW SERIES HELPER Gaps remain disconnected. ========================= */ function drawSeries( key, color, lineWidth ){ ctx.strokeStyle = color; ctx.lineWidth = lineWidth; ctx.lineJoin = "round"; ctx.lineCap = "round"; ctx.beginPath(); let previousValid = false; history.forEach((row,index) => { const value = row[key]; if(!Number.isFinite(value)){ previousValid = false; return; } if(!previousValid){ ctx.moveTo( x(index), y(value) ); } else{ ctx.lineTo( x(index), y(value) ); } previousValid = true; }); ctx.stroke(); } /* ========================= HISTORY LINES ========================= */ ctx.save(); ctx.beginPath(); ctx.rect( pad.l, pad.t, plotW, plotH ); ctx.clip(); /* Benchmark first, portfolio on top. */ drawSeries( "benchmark", "#64748b", 1.5 ); drawSeries( "portfolio", "#4E79BD", 2 ); ctx.restore(); /* ========================= LEGEND ========================= */ ctx.font = '10px "Times New Roman"'; ctx.textAlign = "left"; // Portfolio ctx.strokeStyle = "#4E79BD"; ctx.lineWidth = 2; ctx.beginPath(); ctx.moveTo( pad.l, 16 ); ctx.lineTo( pad.l + 18, 16 ); ctx.stroke(); ctx.fillStyle = "#243b5a"; ctx.fillText( "Portfolio", pad.l + 24, 19 ); // Benchmark ctx.strokeStyle = "#64748b"; ctx.lineWidth = 1.5; ctx.beginPath(); ctx.moveTo( pad.l + 105, 16 ); ctx.lineTo( pad.l + 123, 16 ); ctx.stroke(); ctx.fillStyle = "#243b5a"; ctx.fillText( "Benchmark", pad.l + 129, 19 ); /* ========================= X AXIS DATES ========================= */ const firstDate = history[0]?.date || ""; const lastDate = history[ history.length - 1 ]?.date || ""; ctx.fillStyle = "#64748b"; ctx.font = '10px "Times New Roman"'; ctx.textAlign = "left"; ctx.fillText( firstDate, pad.l, h - 8 ); ctx.textAlign = "right"; ctx.fillText( lastDate, w - pad.r, h - 8 ); /* ========================================= HISTORY TOOLTIP + MOUSE EVENTS ========================================= */ const tooltip = document.getElementById("historyTooltip"); if(!tooltip) return; /* Preserve the completed chart. Mouse movement redraws the chart from this static image. */ const chartImage = document.createElement("canvas"); chartImage.width = canvas.width; chartImage.height = canvas.height; chartImage .getContext("2d") .drawImage(canvas,0,0); /* FORMAT INDEX VALUES */ const formatIndex = value => Number.isFinite(value) ? value.toLocaleString( "en-GB", { minimumFractionDigits:2, maximumFractionDigits:2 } ) : "—"; /* MOUSE MOVE */ canvas.onmousemove = event => { const rect = canvas.getBoundingClientRect(); const mouseX = (event.clientX - rect.left) * w / rect.width; /* Outside plotting area */ if( mouseX < pad.l || mouseX > w - pad.r ){ tooltip.hidden = true; ctx.clearRect(0,0,w,h); ctx.drawImage( chartImage, 0,0,w,h ); return; } /* Nearest monthly observation */ const index = Math.max( 0, Math.min( history.length - 1, Math.round( (mouseX - pad.l) / plotW * (history.length - 1) ) ) ); const row = history[index]; const portfolio = row.portfolio; const benchmark = row.benchmark; const xx = x(index); /* Restore original chart */ ctx.clearRect(0,0,w,h); ctx.drawImage( chartImage, 0,0,w,h ); /* Vertical crosshair */ ctx.save(); ctx.beginPath(); ctx.rect( pad.l, pad.t, plotW, plotH ); ctx.clip(); ctx.strokeStyle = "rgba(78,121,189,.45)"; ctx.lineWidth = 1; ctx.setLineDash([3,3]); ctx.beginPath(); ctx.moveTo( xx, pad.t ); ctx.lineTo( xx, pad.t + plotH ); ctx.stroke(); ctx.setLineDash([]); /* Intersection dots */ const paintDot = ( value, color ) => { if(!Number.isFinite(value)) return; ctx.beginPath(); ctx.arc( xx, y(value), 3.5, 0, Math.PI * 2 ); ctx.fillStyle = "#fff"; ctx.fill(); ctx.strokeStyle = color; ctx.lineWidth = 2; ctx.stroke(); }; paintDot( portfolio, "#4E79BD" ); paintDot( benchmark, "#64748b" ); ctx.restore(); /* Relative cumulative performance Portfolio / Benchmark - 1 */ const difference = Number.isFinite(portfolio) && Number.isFinite(benchmark) && benchmark > 0 ? ( portfolio / benchmark - 1 ) * 100 : NaN; const differenceText = Number.isFinite(difference) ? ( difference > 0 ? "+" : "" ) + difference.toFixed(2) + "%" : "—"; const differenceColor = !Number.isFinite(difference) ? "#64748b" : difference > 0 ? "#4E79BD" : difference < 0 ? "#c24141" : "#64748b"; const isDA = INIT.lang === "da"; const portfolioLabel = isDA ? "Portefølje" : "Portfolio"; const benchmarkLabel = isDA ? "Benchmark" : "Benchmark"; const relativeLabel = isDA ? "Relativt afkast" : "Relative performance"; /* Tooltip content */ tooltip.innerHTML = `
${row.date}
${portfolioLabel} ${formatIndex(portfolio)}
${benchmarkLabel} ${formatIndex(benchmark)}
${relativeLabel} ${differenceText}
`; /* Position inside chart container */ tooltip.hidden = false; const tooltipWidth = tooltip.offsetWidth; const tooltipHeight = tooltip.offsetHeight; const canvasLeft = canvas.offsetLeft; const canvasTop = canvas.offsetTop; const desiredLeft = canvasLeft + xx + 14; const desiredTop = canvasTop + 12; tooltip.style.left = Math.max( 4, Math.min( desiredLeft, box.clientWidth - tooltipWidth - 4 ) ) + "px"; tooltip.style.top = Math.max( 4, Math.min( desiredTop, box.clientHeight - tooltipHeight - 4 ) ) + "px"; }; /* MOUSE LEAVE */ canvas.onmouseleave = () => { tooltip.hidden = true; ctx.clearRect( 0,0,w,h ); ctx.drawImage( chartImage, 0,0,w,h ); }; function setPortfolioDisplayPhase(phase){ if( phase !== "build" && phase !== "use" ){ return; } portfolioViewPhase = phase; document .getElementById("btnPortfolioBuild") ?.classList.toggle( "active", phase === "build" ); document .getElementById("btnPortfolioUse") ?.classList.toggle( "active", phase === "use" ); renderRiskModel(); } }
